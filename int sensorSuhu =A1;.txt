int sensorSuhu =A1;
#define alarm 10
#include<Servo.h>
Servo Servol;
float Posisi;

void setup()
{
  Serial.begin (9600);
  pinMode(alarm, OUTPUT);
  Servol.attach(11);
}

void loop()
{
  int read = analogRead(sensorSuhu);
  float voltage = read * 5.0;
  voltage /=1024.0;
  
  float temperatur = (voltage-0.5)*100;
  Serial.print(temperatur);
  Serial.println(" Derajat Celcius");
  
  if(temperatur > 80)
  {
    tone (alarm, 15);
    Servol.write(180);
    Posisi = 180.0;
  }
  else
  {
    noTone (alarm);
    Servol.write(0);
    Posisi = 0.0;
  }
  delay(1000);
  
  Serial.print(Posisi);
  Serial.println(" degrees");
}
